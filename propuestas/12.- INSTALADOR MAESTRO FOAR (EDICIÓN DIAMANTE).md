## 💎 INSTALADOR MAESTRO FOAR (EDICIÓN DIAMANTE)

Este es el script final. Cópielo y ejecútelo íntegramente en su entorno de Producción.

```sql
-- ============================================================================
-- FRAMEWORK CORPORATIVO DE ORQUESTACIÓN ASÍNCRONA RESILIENTE (FOAR) 
-- EDICIÓN: Diamante (Producción Definitiva)
-- ARQUITECTURA: Procedure-Based + RAM Arrays + Zero-Subtransaction Lock
-- ============================================================================

BEGIN;

-- ----------------------------------------------------------------------------
-- FASE 1: Extensiones, Esquema y Enums
-- ----------------------------------------------------------------------------
CREATE EXTENSION IF NOT EXISTS pg_background;
CREATE EXTENSION IF NOT EXISTS pgcrypto;
CREATE SCHEMA IF NOT EXISTS resilient_bg;

DO $$ 
BEGIN
    IF NOT EXISTS (SELECT 1 FROM pg_type WHERE typname = 'execution_mode' AND typnamespace = 'resilient_bg'::regnamespace) THEN
        CREATE TYPE resilient_bg.execution_mode AS ENUM ('SEQUENTIAL_STRICT', 'SEQUENTIAL_NORMAL', 'PARALLEL_INITIAL', 'RANDOM');
    END IF;
    IF NOT EXISTS (SELECT 1 FROM pg_type WHERE typname = 'run_status' AND typnamespace = 'resilient_bg'::regnamespace) THEN
        CREATE TYPE resilient_bg.run_status AS ENUM ('INITIALIZING', 'RUNNING', 'COMPLETED', 'FAILED', 'RECOVERING');
    END IF;
    IF NOT EXISTS (SELECT 1 FROM pg_type WHERE typname = 'task_status' AND typnamespace = 'resilient_bg'::regnamespace) THEN
        CREATE TYPE resilient_bg.task_status AS ENUM ('PENDING', 'RUNNING', 'SUCCESS', 'FAILED', 'KILLED');
    END IF;
END $$;

-- ----------------------------------------------------------------------------
-- FASE 2: Persistencia Inmutable y Desacoplada
-- ----------------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS resilient_bg.cat_queries (
    query_id SERIAL PRIMARY KEY,
    query_hash VARCHAR(64) UNIQUE NOT NULL,
    query_text TEXT NOT NULL,
    created_at TIMESTAMP DEFAULT CLOCK_TIMESTAMP()
);

CREATE TABLE IF NOT EXISTS resilient_bg.def_jobs (
    job_id SERIAL PRIMARY KEY,
    job_name VARCHAR(100) UNIQUE NOT NULL,
    mode resilient_bg.execution_mode NOT NULL,
    max_parallel_processes INT DEFAULT 1 CHECK (max_parallel_processes >= 1),
    timeout_seconds INT NOT NULL DEFAULT 300 CHECK (timeout_seconds > 0),
    max_retries INT NOT NULL DEFAULT 0 CHECK (max_retries >= 0),
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP DEFAULT CLOCK_TIMESTAMP()
);

CREATE TABLE IF NOT EXISTS resilient_bg.def_tasks (
    task_id SERIAL PRIMARY KEY,
    job_id INT NOT NULL REFERENCES resilient_bg.def_jobs(job_id) ON DELETE CASCADE,
    query_id INT NOT NULL REFERENCES resilient_bg.cat_queries(query_id),
    execution_order INT NOT NULL CHECK (execution_order > 0),
    CONSTRAINT unique_task_order_per_job UNIQUE (job_id, execution_order)
);

CREATE TABLE IF NOT EXISTS resilient_bg.run_jobs (
    run_id SERIAL PRIMARY KEY,
    job_id INT NOT NULL REFERENCES resilient_bg.def_jobs(job_id),
    status resilient_bg.run_status DEFAULT 'INITIALIZING',
    monitor_pid INT,
    started_at TIMESTAMP DEFAULT CLOCK_TIMESTAMP(),
    ended_at TIMESTAMP
);

CREATE TABLE IF NOT EXISTS resilient_bg.run_tasks (
    run_task_id SERIAL PRIMARY KEY,
    run_id INT NOT NULL REFERENCES resilient_bg.run_jobs(run_id) ON DELETE CASCADE,
    query_id INT NOT NULL REFERENCES resilient_bg.cat_queries(query_id),
    execution_order INT NOT NULL,
    status resilient_bg.task_status DEFAULT 'PENDING',
    child_pid INT,
    attempt INT DEFAULT 1,
    started_at TIMESTAMP,
    ended_at TIMESTAMP,
    error_log TEXT
);

CREATE INDEX IF NOT EXISTS idx_cat_queries_hash ON resilient_bg.cat_queries(query_hash);
CREATE INDEX IF NOT EXISTS idx_run_tasks_lookup ON resilient_bg.run_tasks(run_id, status);

-- ----------------------------------------------------------------------------
-- FASE 3: Motores Base de Concurrencia (Lock-Free & Subtransaction-Free)
-- ----------------------------------------------------------------------------
CREATE OR REPLACE FUNCTION resilient_bg.register_query(p_sql TEXT) RETURNS INT AS $$
DECLARE v_id INT;
BEGIN
    INSERT INTO resilient_bg.cat_queries (query_hash, query_text) 
    VALUES (encode(digest(p_sql, 'sha256'), 'hex'), p_sql)
    ON CONFLICT (query_hash) DO UPDATE SET query_hash = EXCLUDED.query_hash 
    RETURNING query_id INTO v_id;
    RETURN v_id;
END;
$$ LANGUAGE plpgsql;

-- [HIJO]: Procedure Aislado. Commits fuera de las Subtransacciones de Excepción.
CREATE OR REPLACE PROCEDURE resilient_bg.bg_task_executor(p_run_task_id INT) AS $$
DECLARE 
    v_query_text TEXT;
BEGIN
    SELECT cq.query_text INTO v_query_text 
    FROM resilient_bg.run_tasks rt JOIN resilient_bg.cat_queries cq ON rt.query_id = cq.query_id 
    WHERE rt.run_task_id = p_run_task_id;

    UPDATE resilient_bg.run_tasks SET status = 'RUNNING', started_at = CLOCK_TIMESTAMP(), child_pid = pg_backend_pid() WHERE run_task_id = p_run_task_id;
    COMMIT; -- Primer guardado seguro (Aislado)

    BEGIN
        EXECUTE v_query_text; 
        UPDATE resilient_bg.run_tasks SET status = 'SUCCESS', ended_at = CLOCK_TIMESTAMP() WHERE run_task_id = p_run_task_id;
    EXCEPTION WHEN OTHERS THEN
        UPDATE resilient_bg.run_tasks SET status = 'FAILED', ended_at = CLOCK_TIMESTAMP(), error_log = SQLERRM WHERE run_task_id = p_run_task_id;
    END;
    
    COMMIT; -- Segundo guardado seguro (Aislado)
END;
$$ LANGUAGE plpgsql;

-- [PADRE]: Orquestador en RAM. Evita Cursores Abiertos y permite Commits Continuos.
CREATE OR REPLACE PROCEDURE resilient_bg.bg_job_orchestrator(p_run_id INT) AS $$
DECLARE
    v_mode resilient_bg.execution_mode; v_timeout INT; v_max_retries INT; v_max_parallel INT;
    v_child_pid INT; v_current_status resilient_bg.task_status; v_curr_attempt INT;
    v_task_start TIMESTAMP; v_active_slots INT; v_run_task_id INT;
    v_task_list INT[]; v_running_list INT[]; v_failed_list INT[]; v_task_started_at TIMESTAMP;
BEGIN
    UPDATE resilient_bg.run_jobs SET monitor_pid = pg_backend_pid(), status = 'RUNNING', started_at = CLOCK_TIMESTAMP() WHERE run_id = p_run_id;
    COMMIT;

    SELECT dj.mode, dj.timeout_seconds, dj.max_retries, dj.max_parallel_processes 
    INTO v_mode, v_timeout, v_max_retries, v_max_parallel
    FROM resilient_bg.run_jobs rj JOIN resilient_bg.def_jobs dj ON rj.job_id = dj.job_id WHERE rj.run_id = p_run_id;

    IF v_mode = 'PARALLEL_INITIAL' THEN
        SELECT COUNT(*) INTO v_max_parallel FROM resilient_bg.run_tasks WHERE run_id = p_run_id;
    END IF;

    -- LÓGICA 1: SECUENCIAL
    IF v_mode IN ('SEQUENTIAL_STRICT', 'SEQUENTIAL_NORMAL') THEN
        v_task_list := ARRAY(SELECT run_task_id FROM resilient_bg.run_tasks WHERE run_id = p_run_id ORDER BY execution_order ASC);
        
        FOREACH v_run_task_id IN ARRAY v_task_list LOOP
            WHILE TRUE LOOP
                SELECT pg_background_launch(format('CALL resilient_bg.bg_task_executor(%L)', v_run_task_id)) INTO v_child_pid;
                v_task_start := CLOCK_TIMESTAMP(); 
                
                WHILE TRUE LOOP
                    COMMIT; 
                    PERFORM pg_sleep(0.5); 
                    
                    SELECT status INTO v_current_status FROM resilient_bg.run_tasks WHERE run_task_id = v_run_task_id;
                    IF v_current_status IN ('SUCCESS', 'FAILED') THEN EXIT; END IF;

                    IF EXTRACT(EPOCH FROM (CLOCK_TIMESTAMP() - v_task_start)) >= v_timeout THEN
                        PERFORM pg_cancel_backend(v_child_pid); 
                        UPDATE resilient_bg.run_tasks SET status = 'FAILED', ended_at = CLOCK_TIMESTAMP(), error_log = 'Aniquilado por el Padre (Timeout estricto)' WHERE run_task_id = v_run_task_id;
                        COMMIT; v_current_status := 'FAILED'; EXIT;
                    END IF;

                    IF NOT EXISTS (SELECT 1 FROM pg_stat_activity WHERE pid = v_child_pid AND backend_type = 'pg_background') THEN
                        UPDATE resilient_bg.run_tasks SET status = 'FAILED', ended_at = CLOCK_TIMESTAMP(), error_log = 'Worker abortado por S.O.' WHERE run_task_id = v_run_task_id;
                        COMMIT; v_current_status := 'FAILED'; EXIT;
                    END IF;
                END LOOP;

                IF v_current_status = 'SUCCESS' THEN EXIT; END IF;

                SELECT attempt INTO v_curr_attempt FROM resilient_bg.run_tasks WHERE run_task_id = v_run_task_id;
                IF v_curr_attempt <= v_max_retries THEN
                    UPDATE resilient_bg.run_tasks SET attempt = v_curr_attempt + 1, status = 'PENDING', error_log = NULL WHERE run_task_id = v_run_task_id;
                    COMMIT;
                ELSE
                    IF v_mode = 'SEQUENTIAL_STRICT' THEN
                        UPDATE resilient_bg.run_jobs SET status = 'FAILED', ended_at = CLOCK_TIMESTAMP() WHERE run_id = p_run_id;
                        COMMIT; RETURN;
                    END IF;
                    EXIT; 
                END IF;
            END LOOP;
        END LOOP;

    -- LÓGICA 2: CONCURRENTE (Random / Parallel Initial)
    ELSE
        WHILE EXISTS (SELECT 1 FROM resilient_bg.run_tasks WHERE run_id = p_run_id AND status IN ('PENDING', 'RUNNING')) LOOP
            COMMIT; 
            
            v_running_list := ARRAY(SELECT run_task_id FROM resilient_bg.run_tasks WHERE run_id = p_run_id AND status = 'RUNNING');
            
            FOREACH v_run_task_id IN ARRAY v_running_list LOOP
                SELECT child_pid, started_at INTO v_child_pid, v_task_started_at FROM resilient_bg.run_tasks WHERE run_task_id = v_run_task_id;

                IF EXTRACT(EPOCH FROM (CLOCK_TIMESTAMP() - v_task_started_at)) >= v_timeout THEN
                    PERFORM pg_cancel_backend(v_child_pid);
                    UPDATE resilient_bg.run_tasks SET status = 'FAILED', ended_at = CLOCK_TIMESTAMP(), error_log = 'Aniquilado por el Padre (Timeout concurrente)' WHERE run_task_id = v_run_task_id;
                    COMMIT;
                ELSIF NOT EXISTS (SELECT 1 FROM pg_stat_activity WHERE pid = v_child_pid AND backend_type = 'pg_background') THEN
                    UPDATE resilient_bg.run_tasks SET status = 'FAILED', ended_at = CLOCK_TIMESTAMP(), error_log = 'Worker concurrente abortado por S.O.' WHERE run_task_id = v_run_task_id;
                    COMMIT;
                END IF;
            END LOOP;

            v_failed_list := ARRAY(SELECT run_task_id FROM resilient_bg.run_tasks WHERE run_id = p_run_id AND status = 'FAILED' AND error_log IS NOT NULL);
            
            FOREACH v_run_task_id IN ARRAY v_failed_list LOOP
                SELECT attempt INTO v_curr_attempt FROM resilient_bg.run_tasks WHERE run_task_id = v_run_task_id;
                IF v_curr_attempt <= v_max_retries THEN
                    UPDATE resilient_bg.run_tasks SET attempt = v_curr_attempt + 1, status = 'PENDING', error_log = NULL WHERE run_task_id = v_run_task_id;
                    COMMIT;
                END IF;
            END LOOP;

            SELECT COUNT(*) INTO v_active_slots FROM resilient_bg.run_tasks WHERE run_id = p_run_id AND status = 'RUNNING';
            
            WHILE v_active_slots < v_max_parallel LOOP
                v_run_task_id := NULL; 
                SELECT run_task_id INTO v_run_task_id FROM resilient_bg.run_tasks WHERE run_id = p_run_id AND status = 'PENDING' ORDER BY execution_order ASC LIMIT 1;
                EXIT WHEN v_run_task_id IS NULL; 

                SELECT pg_background_launch(format('CALL resilient_bg.bg_task_executor(%L)', v_run_task_id)) INTO v_child_pid;
                UPDATE resilient_bg.run_tasks SET status = 'RUNNING', started_at = CLOCK_TIMESTAMP(), child_pid = v_child_pid WHERE run_task_id = v_run_task_id;
                COMMIT;
                
                v_active_slots := v_active_slots + 1;
            END LOOP;

            PERFORM pg_sleep(0.5); 
        END LOOP;
    END IF;
    
    IF EXISTS (SELECT 1 FROM resilient_bg.run_tasks WHERE run_id = p_run_id AND status = 'FAILED') THEN
        UPDATE resilient_bg.run_jobs SET status = 'FAILED', ended_at = CLOCK_TIMESTAMP() WHERE run_id = p_run_id;
    ELSE
        UPDATE resilient_bg.run_jobs SET status = 'COMPLETED', ended_at = CLOCK_TIMESTAMP() WHERE run_id = p_run_id;
    END IF;
    COMMIT;
END;
$$ LANGUAGE plpgsql;

-- ----------------------------------------------------------------------------
-- FASE 4: API de Alto Rendimiento (Modificada con UPSERT y CTE)
-- ----------------------------------------------------------------------------
CREATE OR REPLACE FUNCTION resilient_bg.create_job_definition(
    p_job_name VARCHAR(100), p_mode resilient_bg.execution_mode, p_queries TEXT[], p_timeout_seconds INT DEFAULT 300, p_max_retries INT DEFAULT 0, p_max_parallel_processes INT DEFAULT 1
) RETURNS INT AS $$
DECLARE v_job_id INT; v_query_id INT; v_query_text TEXT; v_order INT := 1;
BEGIN
    INSERT INTO resilient_bg.def_jobs (job_name, mode, timeout_seconds, max_retries, max_parallel_processes)
    VALUES (p_job_name, p_mode, p_timeout_seconds, p_max_retries, p_max_parallel_processes)
    ON CONFLICT (job_name) DO UPDATE 
    SET mode = EXCLUDED.mode, timeout_seconds = EXCLUDED.timeout_seconds, max_retries = EXCLUDED.max_retries, max_parallel_processes = EXCLUDED.max_parallel_processes
    RETURNING job_id INTO v_job_id;

    DELETE FROM resilient_bg.def_tasks WHERE job_id = v_job_id; 

    FOREACH v_query_text IN ARRAY p_queries LOOP
        v_query_id := resilient_bg.register_query(v_query_text);
        INSERT INTO resilient_bg.def_tasks (job_id, query_id, execution_order) VALUES (v_job_id, v_query_id, v_order);
        v_order := v_order + 1;
    END LOOP;
    RETURN v_job_id;
END;
$$ LANGUAGE plpgsql;

CREATE OR REPLACE FUNCTION resilient_bg.start_job(p_job_id INT) RETURNS INT AS $$
DECLARE v_run_id INT; v_mode resilient_bg.execution_mode;
BEGIN
    WITH new_run AS (
        INSERT INTO resilient_bg.run_jobs (job_id, status, started_at) VALUES (p_job_id, 'INITIALIZING', CLOCK_TIMESTAMP()) RETURNING run_id
    ), inserted_tasks AS (
        INSERT INTO resilient_bg.run_tasks (run_id, query_id, execution_order, status)
        SELECT nr.run_id, dt.query_id, dt.execution_order, 'PENDING' FROM resilient_bg.def_tasks dt CROSS JOIN new_run nr WHERE dt.job_id = p_job_id
    )
    SELECT run_id INTO v_run_id FROM new_run;

    PERFORM pg_background_launch(format('CALL resilient_bg.bg_job_orchestrator(%L)', v_run_id));
    
    RETURN v_run_id;
END;
$$ LANGUAGE plpgsql;

CREATE OR REPLACE FUNCTION resilient_bg.launch_job_by_name(p_job_name VARCHAR(100)) RETURNS INT AS $$
DECLARE v_job_id INT;
BEGIN
    SELECT job_id INTO v_job_id FROM resilient_bg.def_jobs WHERE job_name = p_job_name;
    IF v_job_id IS NULL THEN RAISE EXCEPTION 'Job corporativo no encontrado.'; END IF;
    RETURN resilient_bg.start_job(v_job_id);
END;
$$ LANGUAGE plpgsql;

CREATE OR REPLACE FUNCTION resilient_bg.launch_job_one_shot(
    p_job_name VARCHAR(100), p_mode resilient_bg.execution_mode, p_queries TEXT[], p_timeout_seconds INT DEFAULT 300, p_max_retries INT DEFAULT 0, p_max_parallel_processes INT DEFAULT 1
) RETURNS INT AS $$
DECLARE v_job_id INT;
BEGIN
    v_job_id := resilient_bg.create_job_definition(p_job_name, p_mode, p_queries, p_timeout_seconds, p_max_retries, p_max_parallel_processes);
    RETURN resilient_bg.start_job(v_job_id);
END;
$$ LANGUAGE plpgsql;

-- ----------------------------------------------------------------------------
-- FASE 5: Vista de Analítica Pre-Saneada
-- ----------------------------------------------------------------------------
CREATE OR REPLACE VIEW resilient_bg.vw_status_progreso_corporativo AS
WITH cte_metricas AS (
    SELECT rt.run_id, COUNT(*) AS total, COUNT(*) FILTER (WHERE rt.status = 'SUCCESS') AS hechas, COUNT(*) FILTER (WHERE rt.status IN ('FAILED', 'KILLED')) AS errores, COUNT(*) FILTER (WHERE rt.status = 'PENDING') AS espera, COUNT(DISTINCT psa.pid) AS activos
    FROM resilient_bg.run_tasks rt
    LEFT JOIN pg_stat_activity psa ON rt.child_pid = psa.pid AND psa.backend_type = 'pg_background'
    GROUP BY rt.run_id
),
cte_clean AS (
    SELECT rj.run_id, rj.status, rj.started_at, rj.ended_at, dj.job_name, dj.mode,
           COALESCE(m.total, 0) AS total, COALESCE(m.hechas, 0) AS hechas, COALESCE(m.errores, 0) AS errores, COALESCE(m.espera, 0) AS espera, COALESCE(m.activos, 0) AS activos
    FROM resilient_bg.run_jobs rj
    JOIN resilient_bg.def_jobs dj ON rj.job_id = dj.job_id
    LEFT JOIN cte_metricas m ON rj.run_id = m.run_id
)
SELECT 
    run_id AS "ID Ejecución", job_name AS "Nombre del Job", mode AS "Modo de Ejecución",
    CASE 
        WHEN status = 'FAILED' OR (errores > 0 AND mode = 'SEQUENTIAL_STRICT') THEN '❌ FALLIDO CON ERRORES'
        WHEN total > 0 AND total = (hechas + errores) THEN '✅ FINALIZADO'
        WHEN activos > 0 THEN '🔥 EJECUTANDO (MOTOR ACTIVO)'
        WHEN status = 'RECOVERING' THEN '🛡️ EN AUTO-RECUPERACIÓN'
        ELSE '⏳ EN ESPERA / INICIANDO'
    END AS "Estatus Real",
    total AS "Total Tareas", hechas AS "Hechas", errores AS "Errores", espera AS "En Espera", activos AS "Workers Activos (Motor)",
    DATE_TRUNC('second', COALESCE(ended_at, CLOCK_TIMESTAMP()) - started_at) AS "Duración",
    CASE WHEN total = 0 THEN '0%' ELSE ROUND((hechas::FLOAT / total::FLOAT) * 100)::TEXT || '%' END AS "Avance %",
    '[' || REPEAT('█', CASE WHEN total = 0 THEN 0 ELSE ROUND((hechas::FLOAT / total::FLOAT) * 20)::INT END) || 
    REPEAT('░', 20 - CASE WHEN total = 0 THEN 0 ELSE ROUND((hechas::FLOAT / total::FLOAT) * 20)::INT END) || ']' AS "Línea de Progreso"
FROM cte_clean
ORDER BY run_id DESC;



-- ============================================================================
-- COMPONENTE: Multiplicador de Consultas Avanzado (Deduplicación Sintáctica)
-- OPTIMIZACIÓN: Asignación atómica en memoria RAM en un solo paso C (O(1))
-- ============================================================================

CREATE OR REPLACE FUNCTION resilient_bg.replicate_query(
    p_query TEXT, 
    p_times INT
) RETURNS TEXT[] AS $$
BEGIN
    -- Validación defensiva perimetral
    IF p_times IS NULL OR p_times <= 0 THEN
        RAISE EXCEPTION 'Error de Infraestructura: La cantidad de repeticiones debe ser mayor a cero (0).';
    END IF;

    -- [Maestro Turing]: array_fill evita bucles plpgsql y aloja el bloque en memoria instantáneamente
    RETURN array_fill(p_query, ARRAY[p_times]);
END;
$$ LANGUAGE plpgsql IMMUTABLE; 
-- IMMUTABLE le dice al motor que el resultado es determinista, permitiendo caché AST.




COMMIT;
```
